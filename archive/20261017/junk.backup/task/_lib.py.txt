#!/usr/bin/env python3

import os
import json
import shutil
import hashlib
import mimetypes

from datetime import (
    datetime,
    timezone,
    timedelta
)

BASE_DIR = os.path.dirname(
    os.path.abspath(__file__)
)

DATA_FILE = os.path.join(
    BASE_DIR,
    "data.json"
)

REGISTRY_FILE = os.path.join(
    BASE_DIR,
    "registry.json"
)

ATTACHMENTS_DIR = os.path.join(
    BASE_DIR,
    "attachments"
)


def ensure_structure():

    os.makedirs(
        ATTACHMENTS_DIR,
        exist_ok=True
    )

    if not os.path.exists(
        DATA_FILE
    ):

        with open(
            DATA_FILE,
            "w",
            encoding="utf-8"
        ) as f:

            json.dump(
                {"tasks": []},
                f,
                indent=4,
                ensure_ascii=False
            )

    if not os.path.exists(
        REGISTRY_FILE
    ):

        with open(
            REGISTRY_FILE,
            "w",
            encoding="utf-8"
        ) as f:

            json.dump(
                {
                    "last_registry": 0,
                    "last_task": 0
                },
                f,
                indent=4
            )


def load_json(path):

    with open(
        path,
        "r",
        encoding="utf-8"
    ) as f:

        return json.load(f)


def save_json(path, data):

    with open(
        path,
        "w",
        encoding="utf-8"
    ) as f:

        json.dump(
            data,
            f,
            indent=4,
            ensure_ascii=False
        )


def get_time_info():

    utc_now = datetime.now(
        timezone.utc
    )

    dhaka_tz = timezone(
        timedelta(hours=6)
    )

    dhaka_now = utc_now.astimezone(
        dhaka_tz
    )

    return {

        "dhaka":
        dhaka_now.strftime(
            "%Y-%m-%d %H:%M:%S.%f"
        )[:-3],

        "utc":
        utc_now.strftime(
            "%Y-%m-%d %H:%M:%S.%f"
        )[:-3]
    }


def generate_ids():

    registry = load_json(
        REGISTRY_FILE
    )

    registry["last_registry"] += 1
    registry["last_task"] += 1

    reg_id = (
        f"REG-{registry['last_registry']:06}"
    )

    task_id = (
        f"TASK-{registry['last_task']:06}"
    )

    save_json(
        REGISTRY_FILE,
        registry
    )

    return task_id, reg_id


def sha256_file(path):

    digest = hashlib.sha256()

    with open(
        path,
        "rb"
    ) as f:

        for chunk in iter(
            lambda: f.read(4096),
            b""
        ):

            digest.update(chunk)

    return digest.hexdigest()


def store_attachment(
    task_id,
    file_path
):

    if not os.path.exists(
        file_path
    ):
        return None

    filename = os.path.basename(
        file_path
    )

    ext = os.path.splitext(
        filename
    )[1]

    save_name = (
        task_id + ext
    )

    destination = os.path.join(
        ATTACHMENTS_DIR,
        save_name
    )

    shutil.copy2(
        file_path,
        destination
    )

    return {

        "file":
        f"attachments/{save_name}",

        "original":
        filename,

        "size":
        os.path.getsize(
            destination
        ),

        "mime":
        (
            mimetypes.guess_type(
                destination
            )[0]
            or
            "unknown"
        ),

        "sha256":
        sha256_file(
            destination
        )
    }


def create_steps(
    step_string
):

    result = []

    if not step_string:

        return result

    items = [

        x.strip()

        for x in step_string.split(",")

        if x.strip()
    ]

    index = 1

    for item in items:

        result.append({

            "id":
            index,

            "title":
            item,

            "completed":
            False,

            "completed_dhaka":
            None,

            "completed_utc":
            None
        })

        index += 1

    return result
