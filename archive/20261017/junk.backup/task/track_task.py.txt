#!/usr/bin/env python3

import sys

from _lib import (

    DATA_FILE,

    load_json,

    save_json,

    get_time_info,

    find_task,

    calculate_progress
)


def main():

    if len(sys.argv) != 3:

        print()

        print(
            "Usage:"
        )

        print()

        print(
            "python3 track_task.py TASK-000001 1"
        )

        print()

        return

    task_id = sys.argv[1]

    step_id = int(
        sys.argv[2]
    )

    data = load_json(
        DATA_FILE
    )

    task = find_task(
        task_id,
        data
    )

    if not task:

        print(
            f"Task not found: {task_id}"
        )

        return

    target = None

    for step in task["steps"]:

        if (
            step["id"] ==
            step_id
        ):

            target = step
            break

    if not target:

        print(
            f"Step not found: {step_id}"
        )

        return

    if target["completed"]:

        print()

        print(
            "Step already completed."
        )

        print()

        return

    time_info = get_time_info()

    target[
        "completed"
    ] = True

    target[
        "completed_dhaka"
    ] = time_info["dhaka"]

    target[
        "completed_utc"
    ] = time_info["utc"]

    calculate_progress(
        task
    )

    save_json(
        DATA_FILE,
        data
    )

    print()

    print(
        "Step Completed"
    )

    print()

    print(
        f"Task : {task_id}"
    )

    print(
        f"Step : {step_id}"
    )

    print(
        f"Title: {target['title']}"
    )

    print()

    print(
        f"Progress: "
        f"{task['progress']['percentage']}%"
    )

    print()


if __name__ == "__main__":

    main()
